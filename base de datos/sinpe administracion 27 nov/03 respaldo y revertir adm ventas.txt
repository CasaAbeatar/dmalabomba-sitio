CREATE OR REPLACE FUNCTION fact.sinpe_adm_ventas(p_soc uuid, p_desde timestamp with time zone, p_hasta timestamp with time zone, p_pruebas boolean DEFAULT false)
 RETURNS TABLE(pago_id uuid, creado_en timestamp with time zone, monto numeric, referencia text, ref_norm text, venta_id uuid, venta_numero text, cobrado_por text, clave text)
 LANGUAGE sql

AS $function$
  select p.id, p.creado_en, p.monto, p.referencia, fact.sinpe_ref_norm(p.referencia), v.id, v.numero, p.cobrado_por, cp.clave
    from fact.pagos p
    join fact.ventas v on v.id = p.venta_id and v.sociedad_id = p_soc and v.anulada_motivo is null

     and (p_pruebas or not (coalesce(p.cobrado_por, '') = 'prueba' or coalesce(p.cobrado_por, '') like 'zz%' or coalesce(p.referencia, '') ~* '^prueba'))
