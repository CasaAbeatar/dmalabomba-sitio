CREATE OR REPLACE FUNCTION public.fx_sinpe_adm_informe_mes(p_soc uuid, p_mes date)
 RETURNS jsonb
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO ''
AS $function$
declare c record; r record; m0 date := date_trunc('month', p_mes)::date; m1 date := (date_trunc('month', p_mes) + interval '1 month' - interval '1 day')::date; v_fac jsonb; v_nc jsonb;
begin
  select * into c from fact.sinpe_adm_ctx(p_soc);
  select * into r from fact.sinpe_adm_rango(m0, m1);
  select jsonb_build_object('n', count(distinct cc.id), 'total', coalesce(sum((e ->> 'monto')::numeric), 0)) into v_fac
    from fact.comprobantes cc cross join lateral jsonb_array_elements(coalesce(cc.pagos_snapshot, '[]'::jsonb)) e
   where cc.sociedad_id = p_soc and cc.tipo_doc in ('01', '04') and cc.estado = 'aceptado' and cc.fecha_emision >= r.d0 and cc.fecha_emision < r.d1
     and e ->> 'medio' = 'sinpe'
     and (c.ve_pruebas or not (coalesce(e ->> 'cobrado_por', '') = 'prueba' or coalesce(e ->> 'cobrado_por', '') like 'zz%' or coalesce(e ->> 'referencia', '') ~* '^prueba'));
  select jsonb_build_object('n', count(*), 'total', coalesce(sum(cc.total), 0)) into v_nc
    from fact.comprobantes cc where cc.sociedad_id = p_soc and cc.tipo_doc = '03' and cc.estado = 'aceptado' and cc.fecha_emision >= r.d0 and cc.fecha_emision < r.d1;
  return jsonb_build_object('empresa', c.emp, 'mes', m0, 'desde', m0, 'hasta', m1, 'medio06_facturas', v_fac, 'notas_credito_mes', v_nc,
    'resumen', (select jsonb_build_object('n', count(*), 'total', coalesce(sum(b.monto), 0),
           'con_venta_n', count(*) filter (where b.venta_id is not null), 'con_venta_total', coalesce(sum(b.monto) filter (where b.venta_id is not null), 0),

           'sin_venta_n', count(*) filter (where b.venta_id is null), 'sin_venta_total', coalesce(sum(b.monto) filter (where b.venta_id is null), 0),
           'revision_n', count(*) filter (where b.estado = 'revision'), 'historial_n', count(*) filter (where b.origen = 'historial'))
         from fact.sinpe_adm_base(p_soc, r.d0, r.d1, c.ve_pruebas) b),
    'ultimo_aviso', (select max(sp.fecha_pago) from fact.sinpe_pagos sp where sp.empresa is not distinct from c.emp and (c.ve_pruebas or not coalesce(sp.es_prueba, false)) and sp.estado <> 'duplicado'),

CREATE OR REPLACE FUNCTION fact.sinpe_adm_base(p_soc uuid, p_desde timestamp with time zone, p_hasta timestamp with time zone, p_pruebas boolean DEFAULT false)
 RETURNS TABLE(id uuid, fecha_pago timestamp with time zone, dia date, hora integer, dow integer, monto numeric, referencia text, estado text, origen text, pagador text, detalle text, venta_id uuid, venta_numero text, via text, cobrado_por text, clave text, tipo_doc text, comprobante_estado text)
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO ''
AS $function$
  select sp.id, sp.fecha_pago,
         (sp.fecha_pago at time zone 'America/Costa_Rica')::date,
         extract(hour from sp.fecha_pago at time zone 'America/Costa_Rica')::int,
         extract(isodow from sp.fecha_pago at time zone 'America/Costa_Rica')::int,
         sp.monto, sp.referencia_banco, sp.estado,
         case when sp.motivo = 'historico' then 'historial' else 'en_vivo' end,
         sp.pagador_nombre_parcial, sp.detalle_pagador,
         lk.venta_id, v.numero, lk.via, cb.cobrado_por, cp.clave, cp.tipo_doc, cp.estado
    from fact.sinpe_pagos sp
    join fact.sociedades so on so.id = p_soc and sp.empresa is not distinct from so.codigo
    left join lateral (
      select x.venta_id, x.via from (
        select sp.venta_id as venta_id, 1 as o, 'asignada'::text as via where sp.venta_id is not null
        union all
        select u.venta_id, 2, 'marcada'::text from fact.sinpe_usados u where u.pago_id = sp.id and u.venta_id is not null
        union all
        select p2.venta_id, 3, 'referencia'::text from fact.pagos p2 join fact.ventas v2 on v2.id = p2.venta_id
         where p2.medio = 'sinpe' and v2.sociedad_id = p_soc and sp.referencia_banco is not null and fact.sinpe_ref_norm(p2.referencia) = sp.referencia_banco
      ) x order by x.o limit 1) lk on true
    left join fact.ventas v on v.id = lk.venta_id
    left join lateral (select pg.cobrado_por from fact.pagos pg where pg.venta_id = v.id and pg.medio = 'sinpe' order by pg.creado_en limit 1) cb on true
    left join lateral (select c.clave, c.tipo_doc, c.estado from fact.comprobantes c where c.venta_id = v.id and c.tipo_doc in ('01','04')
                        order by (c.estado = 'aceptado') desc, c.creado_en desc limit 1) cp on true
   where sp.estado <> 'duplicado' and (p_pruebas or not coalesce(sp.es_prueba, false))
     and sp.fecha_pago >= p_desde and sp.fecha_pago < p_hasta
$function$;
